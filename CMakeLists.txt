cmake_minimum_required(VERSION 3.16)
project(tesy_primitives VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(TESY_SANITIZE "Enable address and undefined behavior sanitizers" OFF)
find_package(Threads REQUIRED)
add_library(tesy_expert_store native/expert_store.cpp)
target_include_directories(tesy_expert_store PUBLIC native)
target_link_libraries(tesy_expert_store PUBLIC Threads::Threads)
target_compile_options(tesy_expert_store PRIVATE -Wall -Wextra -Wpedantic -Werror -ffp-contract=off)
add_executable(tesy_native_selftest native/selftest.cpp)
target_link_libraries(tesy_native_selftest PRIVATE tesy_expert_store)
target_compile_options(tesy_native_selftest PRIVATE -Wall -Wextra -Wpedantic -Werror -ffp-contract=off)
if(TESY_SANITIZE)
  foreach(target tesy_expert_store tesy_native_selftest)
    target_compile_options(${target} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(${target} PRIVATE -fsanitize=address,undefined)
  endforeach()
endif()
enable_testing()
add_test(NAME native_f32_residency COMMAND tesy_native_selftest)
