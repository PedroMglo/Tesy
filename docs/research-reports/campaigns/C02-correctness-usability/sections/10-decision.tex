\section{Decisão}
\begin{FindingBox}{CORRECTNESS\_BLOCKED permanece o resultado global}
\claim{CL-C2-STATE}{Preservar os resultados de função, timing e recursos nos seus âmbitos, sem transformar a cobertura incompleta de correção numa aprovação global. Não há demonstração de impossibilidade física universal.}
\end{FindingBox}
\claim{CL-C2-BOUNDARY}{Não promover placement, prefetch ou speculation com base nesta campanha. A ausência de passagem no gate numérico é material.}
\claim{CL-C2-UTILITY}{O fallback seletivo requer um protocolo próprio de qualidade, latência e custo. O piloto permite formular esse teste, mas não o valida em produção.}
