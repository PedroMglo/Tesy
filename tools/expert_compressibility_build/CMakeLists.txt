cmake_minimum_required(VERSION 3.22)
project(tesy_lossless_characterization LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
find_package(OpenSSL 3.5.9 EXACT REQUIRED)
find_library(TESY_ZSTD_LIBRARY NAMES zstd REQUIRED)
add_executable(expert-compressibility "${TESY_REPO}/tools/expert_compressibility.cpp")
target_include_directories(expert-compressibility PRIVATE "${TESY_BACKEND}/vendor/nlohmann")
target_link_libraries(expert-compressibility PRIVATE "${TESY_ZSTD_LIBRARY}" OpenSSL::Crypto)
