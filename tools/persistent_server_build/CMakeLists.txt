# Standalone mature server build. Numerical/common libraries are IMPORTED,
# never rebuilt here. No UI download, new model, runtime or kernel.
cmake_minimum_required(VERSION 3.22)
project(tesy_persistent_server LANGUAGES C CXX)
if(NOT DEFINED TESY_BACKEND OR NOT DEFINED TESY_QUALIFIED_BIN)
  message(FATAL_ERROR "Explicit pinned backend and qualified library directory required")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(BUILD_SHARED_LIBS ON)
set(LLAMA_INSTALL_VERSION 0.0.9884)
set(LLAMA_TOOLS_INSTALL OFF)
set(LLAMA_BUILD_BORINGSSL OFF)
set(LLAMA_BUILD_LIBRESSL OFF)
set(LLAMA_OPENSSL ON)
set(MTMD_VIDEO ON)
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
set(CMAKE_LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
set(CMAKE_BUILD_RPATH "${TESY_QUALIFIED_BIN};${CMAKE_BINARY_DIR}/bin")
set(CMAKE_INSTALL_RPATH "${TESY_QUALIFIED_BIN};${CMAKE_BINARY_DIR}/bin")
function(license_add_file)
endfunction()
function(llama_add_compile_flags)
endfunction()
find_package(Threads REQUIRED)
include_directories("${TESY_BACKEND}" "${TESY_BACKEND}/include" "${TESY_BACKEND}/common" "${TESY_BACKEND}/vendor" "${TESY_BACKEND}/vendor/nlohmann" "${TESY_BACKEND}/ggml/include" "${TESY_BACKEND}/src")
add_compile_definitions(LLAMA_SHARED GGML_SHARED GGML_BACKEND_SHARED GGML_USE_CPU GGML_USE_CUDA CPPHTTPLIB_OPENSSL_SUPPORT)
foreach(lib llama llama-common ggml ggml-base ggml-cpu)
  add_library(${lib} SHARED IMPORTED GLOBAL)
  set_target_properties(${lib} PROPERTIES IMPORTED_LOCATION "${TESY_QUALIFIED_BIN}/lib${lib}.so")
endforeach()
target_link_libraries(llama-common INTERFACE llama ggml Threads::Threads)
target_link_libraries(llama INTERFACE ggml ggml-base)
target_link_libraries(ggml INTERFACE ggml-base)
# Reuse pinned vendored HTTP implementation/system OpenSSL, no fetch.
add_subdirectory("${TESY_BACKEND}/vendor/cpp-httplib" "${CMAKE_BINARY_DIR}/cpp-httplib")
# Reuse original embedding generator with its documented empty-assets mode.
add_executable(tesy-ui-embed "${TESY_BACKEND}/tools/ui/embed.cpp")
add_custom_command(OUTPUT "${CMAKE_BINARY_DIR}/ui.cpp" "${CMAKE_BINARY_DIR}/ui.h" COMMAND tesy-ui-embed "${CMAKE_BINARY_DIR}/ui.cpp" "${CMAKE_BINARY_DIR}/ui.h" DEPENDS tesy-ui-embed VERBATIM)
add_library(llama-ui STATIC "${CMAKE_BINARY_DIR}/ui.cpp" "${CMAKE_BINARY_DIR}/ui.h")
target_include_directories(llama-ui PUBLIC "${CMAKE_BINARY_DIR}")
# Original mature server and its unmodified multimodal link dependency.
add_subdirectory("${TESY_BACKEND}/tools/mtmd" "${CMAKE_BINARY_DIR}/mtmd")
add_subdirectory("${TESY_BACKEND}/tools/server" "${CMAKE_BINARY_DIR}/server")
