cmake_minimum_required(VERSION 3.22)
project(tesy_score_affine LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
set(CMAKE_BUILD_RPATH "${TESY_QUALIFIED_BIN}")
add_compile_definitions(GGML_SHARED GGML_BACKEND_SHARED GGML_USE_CPU)
foreach(lib ggml ggml-base ggml-cpu)
 add_library(${lib} SHARED IMPORTED GLOBAL)
 set_target_properties(${lib} PROPERTIES IMPORTED_LOCATION "${TESY_QUALIFIED_BIN}/lib${lib}.so")
endforeach()
add_executable(expert-score-native "${TESY_REPO}/tools/expert_score_native.cpp")
target_include_directories(expert-score-native PRIVATE "${TESY_BACKEND}/ggml/include" "${TESY_BACKEND}/vendor/nlohmann")
target_link_libraries(expert-score-native PRIVATE ggml ggml-base ggml-cpu)
