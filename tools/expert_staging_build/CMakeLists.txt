cmake_minimum_required(VERSION 3.22)
project(tesy_cpu_staging LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(BUILD_SHARED_LIBS ON)
set(LLAMA_INSTALL_VERSION 0.0.9884)
set(CMAKE_LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
set(CMAKE_BUILD_RPATH "${CMAKE_BINARY_DIR}/bin;${TESY_QUALIFIED_BIN}")
function(llama_add_compile_flags)
endfunction()
find_package(Threads REQUIRED)
add_compile_definitions(LLAMA_SHARED GGML_SHARED GGML_BACKEND_SHARED GGML_USE_CPU GGML_USE_CUDA)
foreach(lib ggml ggml-base ggml-cpu ggml-cuda)
 add_library(${lib} SHARED IMPORTED GLOBAL)
 set_target_properties(${lib} PROPERTIES IMPORTED_LOCATION "${TESY_QUALIFIED_BIN}/lib${lib}.so" INTERFACE_INCLUDE_DIRECTORIES "${TESY_BACKEND}/ggml/include")
endforeach()
target_link_libraries(ggml INTERFACE ggml-base Threads::Threads)
add_subdirectory("${TESY_BACKEND}/src" "${CMAKE_BINARY_DIR}/llama")
target_compile_definitions(llama PRIVATE TESY_C47_CPU_SKIP_PARKED)
foreach(exe expert-prediction-staging expert-staging-fixture)
 string(REPLACE "-" "_" source "${exe}")
 add_executable(${exe} "${TESY_REPO}/tools/${source}.cpp")
 target_include_directories(${exe} PRIVATE "${TESY_BACKEND}/src" "${TESY_BACKEND}/vendor/nlohmann")
 target_link_libraries(${exe} PRIVATE llama ggml ggml-base ggml-cpu ggml-cuda Threads::Threads)
endforeach()
