cmake_minimum_required(VERSION 3.22)
project(tesy_score_corpus LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
set(CMAKE_BUILD_RPATH "${TESY_WINDOW_BIN};${TESY_QUALIFIED_BIN}")
find_package(Threads REQUIRED)
add_compile_definitions(LLAMA_SHARED GGML_SHARED GGML_BACKEND_SHARED GGML_USE_CPU GGML_USE_CUDA)
foreach(lib ggml ggml-base ggml-cpu ggml-cuda)
 add_library(${lib} SHARED IMPORTED GLOBAL)
 set_target_properties(${lib} PROPERTIES IMPORTED_LOCATION "${TESY_QUALIFIED_BIN}/lib${lib}.so")
endforeach()
add_library(llama SHARED IMPORTED GLOBAL)
set_target_properties(llama PROPERTIES IMPORTED_LOCATION "${TESY_WINDOW_BIN}/libllama.so")
add_library(llama-common SHARED IMPORTED GLOBAL)
set_target_properties(llama-common PROPERTIES IMPORTED_LOCATION "${TESY_QUALIFIED_BIN}/libllama-common.so")
add_executable(expert-prediction-corpus "${TESY_REPO}/tools/expert_prediction_corpus.cpp")
target_include_directories(expert-prediction-corpus PRIVATE "${TESY_BACKEND}/include" "${TESY_BACKEND}/ggml/include" "${TESY_BACKEND}/src" "${TESY_BACKEND}/common" "${TESY_BACKEND}/vendor/nlohmann" "${TESY_BACKEND}/vendor")
target_link_libraries(expert-prediction-corpus PRIVATE llama-common llama ggml ggml-base ggml-cpu ggml-cuda Threads::Threads)
