cmake_minimum_required(VERSION 3.22)
project(tesy_logical_route LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(BUILD_SHARED_LIBS ON)
set(LLAMA_INSTALL_VERSION 0.0.9884)
set(CMAKE_LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
set(CMAKE_BUILD_RPATH "${CMAKE_BINARY_DIR}/bin;${TESY_QUALIFIED_BIN}")
function(llama_add_compile_flags)
endfunction()
find_package(Threads REQUIRED)
add_compile_definitions(LLAMA_SHARED GGML_SHARED GGML_BACKEND_SHARED GGML_USE_CPU GGML_USE_CUDA)
foreach(lib ggml ggml-base ggml-cpu ggml-cuda)
 add_library(${lib} SHARED IMPORTED GLOBAL)
 set_target_properties(${lib} PROPERTIES IMPORTED_LOCATION "${TESY_QUALIFIED_BIN}/lib${lib}.so" INTERFACE_INCLUDE_DIRECTORIES "${TESY_BACKEND}/ggml/include")
endforeach()
target_link_libraries(ggml INTERFACE ggml-base Threads::Threads)
add_subdirectory("${TESY_BACKEND}/src" "${CMAKE_BINARY_DIR}/llama")
target_compile_definitions(llama PRIVATE TESY_C47_CPU_SKIP_PARKED)

add_executable(expert-service-native "${TESY_REPO}/tools/expert_service_native.cpp")
target_include_directories(expert-service-native PRIVATE "${TESY_BACKEND}/src" "${TESY_BACKEND}/vendor/nlohmann")
target_link_libraries(expert-service-native PRIVATE llama ggml ggml-base ggml-cpu ggml-cuda Threads::Threads)
add_executable(expert-store-capture "${TESY_REPO}/tools/c127_slots40_boundary_capture.cpp")
target_include_directories(expert-store-capture PRIVATE "${TESY_BACKEND}/src" "${TESY_BACKEND}/vendor/nlohmann")
target_link_libraries(expert-store-capture PRIVATE llama ggml ggml-base ggml-cpu ggml-cuda Threads::Threads)
add_executable(test-expert-store "${TESY_REPO}/tools/test_expert_store.cpp")
target_include_directories(test-expert-store PRIVATE "${TESY_BACKEND}/src" "${TESY_BACKEND}/vendor/nlohmann")
target_compile_options(test-expert-store PRIVATE -UNDEBUG)
target_link_libraries(test-expert-store PRIVATE ggml ggml-base ggml-cpu ggml-cuda Threads::Threads)
