cmake_minimum_required(VERSION 3.22)
project(tesy_logical_route LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(BUILD_SHARED_LIBS ON)
set(LLAMA_INSTALL_VERSION 0.0.9884)
set(CMAKE_LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
set(CMAKE_BUILD_RPATH "${CMAKE_BINARY_DIR}/bin;${TESY_QUALIFIED_BIN}")
function(llama_add_compile_flags)
endfunction()
find_package(Threads REQUIRED)
add_compile_definitions(LLAMA_SHARED GGML_SHARED GGML_BACKEND_SHARED GGML_USE_CPU GGML_USE_CUDA)
foreach(lib ggml ggml-base ggml-cpu ggml-cuda)
 add_library(${lib} SHARED IMPORTED GLOBAL)
 set_target_properties(${lib} PROPERTIES IMPORTED_LOCATION "${TESY_QUALIFIED_BIN}/lib${lib}.so" INTERFACE_INCLUDE_DIRECTORIES "${TESY_BACKEND}/ggml/include")
endforeach()
target_link_libraries(ggml INTERFACE ggml-base Threads::Threads)
add_subdirectory("${TESY_BACKEND}/src" "${CMAKE_BINARY_DIR}/llama")
target_compile_definitions(llama PRIVATE TESY_C47_CPU_SKIP_PARKED)
add_executable(logical-route-native "${TESY_REPO}/tools/logical_route_native.cpp")
target_include_directories(logical-route-native PRIVATE "${TESY_BACKEND}/src" "${TESY_BACKEND}/vendor/nlohmann")
target_link_libraries(logical-route-native PRIVATE llama ggml ggml-base ggml-cpu ggml-cuda Threads::Threads)
add_executable(test-logical-route "${TESY_REPO}/tools/test_logical_route.cpp")
target_include_directories(test-logical-route PRIVATE "${TESY_BACKEND}/src")
target_compile_options(test-logical-route PRIVATE -UNDEBUG)
add_executable(logical-route-capture "${TESY_REPO}/tools/block_layer_reuse_probe.cpp")
target_include_directories(logical-route-capture PRIVATE "${TESY_BACKEND}/src" "${TESY_BACKEND}/vendor/nlohmann")
target_link_libraries(logical-route-capture PRIVATE llama ggml ggml-base ggml-cpu ggml-cuda Threads::Threads)
add_executable(service-gguf-inventory "${TESY_REPO}/tools/c3_gguf_inventory.cpp")
target_link_libraries(service-gguf-inventory PRIVATE ggml-base)
